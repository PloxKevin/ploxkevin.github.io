def infiniteViolationCount (k goal sigma : ℝ) (T : ℕ) (w : ℕ → ℝ) : ℝ :=
  ∑ t : Fin T, stepCost {w | 1 < infiniteTrajectory k goal sigma w (t.val+1)} w

theorem actual_infinite_state_is_measurable (k goal sigma : ℝ) (t : ℕ) :
    Measurable (fun w : ℕ → ℝ => infiniteTrajectory k goal sigma w t) := by
  induction t with
  | zero => exact measurable_const
  | succ t ih =>
    have hn : Measurable (fun w : ℕ → ℝ => sigma*w t) :=
      measurable_const.mul (measurable_pi_apply t)
    exact (ih.sub ((ih.sub measurable_const).const_mul k)).add
      hn

theorem the_printed_standardized_event_is_pointwise_the_original_infinite_state_event
    (k goal sigma : ℝ) (t : ℕ) (hv : 0 < trueVariance k sigma t) (w : ℕ → ℝ) :
    (1 < infiniteTrajectory k goal sigma w t) ↔
      ((1-trueMean k goal t)/Real.sqrt (trueVariance k sigma t) <
        (infiniteTrajectory k goal sigma w t-trueMean k goal t)/Real.sqrt (trueVariance k sigma t)) := by
  simp only [div_lt_div_iff_of_pos_right (Real.sqrt_pos_of_pos hv), sub_lt_sub_iff_right]

theorem actual_standardized_infinite_positive_time_state_has_standard_gaussian_law
    (k goal sigma : ℝ) (t : ℕ) (ht : 0 < t) (hsigma : sigma ≠ 0) :
    HasLaw (fun w : ℕ → ℝ =>
      (infiniteTrajectory k goal sigma w t-trueMean k goal t)/Real.sqrt (trueVariance k sigma t))
      (gaussianReal 0 1) infiniteNoiseLaw := by
  have hl := actual_infinite_positive_time_state_has_the_existing_standard_deviation_law k goal sigma t
  have hn := (actual_positive_time_and_nonzero_noise_make_the_standardizing_denominator_nonzero
    k sigma t ht hsigma).1
  have h := gaussianReal_div_const (gaussianReal_sub_const hl (trueMean k goal t))
    (stateSD k sigma t : ℝ)
  have hv : NNReal.mk ((stateSD k sigma t : ℝ)^2) (sq_nonneg _) = (stateSD k sigma t)^2 := by
    ext
    rfl
  rw [hv, div_self (pow_ne_zero 2 hn)] at h
  norm_num only [sub_self, zero_div] at h
  simpa only [stateSD, Real.sqrt] using h

theorem actual_infinite_one_based_failure_probability_is_the_true_piecewise_tail
    (k goal sigma : ℝ) (T : ℕ) (t : Fin T) :
    infiniteNoiseLaw.real {w | 1 < infiniteTrajectory k goal sigma w (t.val+1)} =
      exactTail k goal sigma (t.val+1) := by
  have hp := actual_infinite_noise_prefix_has_the_existing_finite_product_law T
  have he := hp.measureReal_eq (p := fun w : Fin T → ℝ => 1 < stateAt k goal sigma t w)
    (measurableSet_Ioi.preimage (actual_one_based_state_is_measurable k goal sigma T t))
  have hs : (fun w : ℕ → ℝ => stateAt k goal sigma t (fun j : Fin T => w j.val)) =
      (fun w => infiniteTrajectory k goal sigma w (t.val+1)) := by
    funext w
    exact (actual_infinite_trajectory_agrees_with_every_existing_finite_horizon_prefix
      k goal sigma w T (t.val+1) (by omega)).symm
  have hset : {w : ℕ → ℝ | 1 < stateAt k goal sigma t (fun j : Fin T => w j.val)} =
      {w | 1 < infiniteTrajectory k goal sigma w (t.val+1)} := by
    ext w
    change (1 < stateAt k goal sigma t (fun j : Fin T => w j.val)) ↔ _
    rw [congrFun hs w]
  rw [hset] at he
  exact he.trans (actual_each_one_based_violation_probability_is_the_printed_piecewise_tail
    k goal sigma T t)

theorem actual_infinite_violation_count_is_the_finite_prefix_count_pointwise
    (k goal sigma : ℝ) (T : ℕ) (w : ℕ → ℝ) :
    infiniteViolationCount k goal sigma T w =
      violationCount k goal sigma (fun j : Fin T => w j.val) := by
  classical
  unfold infiniteViolationCount violationCount
  apply Finset.sum_congr rfl
  intro t _
  have hs := actual_infinite_trajectory_agrees_with_every_existing_finite_horizon_prefix
    k goal sigma w T (t.val+1) (by omega)
  simp only [stepCost, Set.indicator_apply, mem_setOf_eq, hs, stateAt,
    vectorTrajectory, oneBasedTime]

      (measurableSet_Ioi.preimage (actual_infinite_state_is_measurable k goal sigma (t.val+1))))]
  apply Finset.sum_congr rfl
  intro t _

    (measurableSet_Ioi.preimage (actual_infinite_state_is_measurable k goal sigma (t.val+1)))
