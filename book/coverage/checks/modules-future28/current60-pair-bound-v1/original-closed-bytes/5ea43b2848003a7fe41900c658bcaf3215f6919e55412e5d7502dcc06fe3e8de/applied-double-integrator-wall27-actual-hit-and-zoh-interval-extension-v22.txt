/-- Continuous actual paths really hit the closed wall by the displayed root;
this is stronger than the previously proved endpoint-in-failure deadline. -/
theorem actual_every_control_reaches_the_wall_by_the_displayed_first_root
    (a L x v : ℝ) (ha : 0 < a) (hx : x < L) (hv : 0 < v)
    (hpeak : L ≤ x+v^2/(2*a)) (P : ControlledPath a x v) :
    let r := (v-Real.sqrt (v^2-2*a*(L-x)))/a
    ∃ t ∈ Icc 0 r, P.position t=L := by
  have hr := actual_braking_first_wall_hit_root a L x v ha hx hv hpeak
  dsimp only at hr ⊢
  have hbound := actual_every_control_fails_by_the_displayed_first_root a L x v ha hx hv hpeak P
  dsimp only at hbound
  have hcontinuous := (P.position_ac
    ((v-Real.sqrt (v^2-2*a*(L-x)))/a) hr.1.le).continuousOn
  rw [uIcc_of_le hr.1.le] at hcontinuous
  exact intermediate_value_Icc hr.1.le hcontinuous
    ⟨by simpa only [P.position_zero] using hx.le,hbound⟩

/-- The actual attained first-hold/constant-tail input is the zero-order-hold
sequence (u,tail,tail,...), on every full half-open sampling interval. -/
theorem actual_first_hold_constant_tail_is_a_zero_order_hold_sequence
    (a x v Delta u tail : ℝ) (hD : 0 < Delta) (P : ControlledPath a x v)
    (hinput : ∀ t : ℝ, P.input t = if t < Delta then u else tail) :
    ∀ (n : ℕ) (t : ℝ), (n:ℝ)*Delta ≤ t → t < ((n+1:ℕ):ℝ)*Delta →
      P.input t = if n=0 then u else tail := by
  intro n t hlo hhi
  cases n with
  | zero =>
      have ht : t < Delta := by simpa only [Nat.cast_zero,Nat.cast_one,one_mul] using hhi
      simp [hinput,ht]
  | succ n =>
      have hprod : 0 ≤ (n:ℝ)*Delta := mul_nonneg (Nat.cast_nonneg n) hD.le
      have hafter : Delta ≤ t := by
        simp only [Nat.cast_add,Nat.cast_one,Nat.succ_eq_add_one] at hlo
        nlinarith
      simp [hinput,not_lt_of_ge hafter]

end SafeLearning.CompleteAppliedDoubleIntegratorWall27
