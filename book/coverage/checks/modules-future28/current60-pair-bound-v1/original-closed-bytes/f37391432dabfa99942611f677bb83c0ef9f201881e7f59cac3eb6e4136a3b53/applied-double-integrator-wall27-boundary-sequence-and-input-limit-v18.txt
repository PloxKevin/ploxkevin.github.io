/-- Every actual safe held-action selector in the fixed-positive-velocity
boundary approach converges to the extreme full-braking input, not just its
Lebesgue measure tending to zero. No regularity or assumed limiting input is
required of the selector. -/
theorem actual_every_braking_boundary_safe_input_selector_tends_to_full_braking
    (a L v Delta : ℝ) (ha : 0 < a) (hv : 0 < v) (hD : 0 < Delta)
    (choose : ℝ → ℝ)
    (hsafe : ∀ epsilon : ℝ, 0 < epsilon →
      choose epsilon ∈ heldContinuationActions a L (L-epsilon-v^2/(2*a)) v Delta) :
    Tendsto choose (𝓝[>] 0) (𝓝 (-a)) := by
  have hup : Tendsto (fun epsilon : ℝ => -a+2*a*epsilon/(v*min Delta (v/(2*a))))
      (𝓝[>] 0) (𝓝 (-a)) := by
    have hi : Tendsto (fun epsilon : ℝ => epsilon) (𝓝[>] 0) (𝓝 0) := nhdsWithin_le_nhds
    convert tendsto_const_nhds.add ((hi.const_mul (2*a)).div_const (v*min Delta (v/(2*a))))
      using 1 <;> simp
  apply tendsto_const_nhds.squeeze' hup
  · filter_upwards [self_mem_nhdsWithin] with epsilon he
    exact (actual_braking_parabola_safe_actions_fit_the_shrinking_interval a L v epsilon Delta ha hv hD
      (hsafe epsilon he)).1
  · filter_upwards [self_mem_nhdsWithin] with epsilon he
    exact (actual_braking_parabola_safe_actions_fit_the_shrinking_interval a L v epsilon Delta ha hv hD
      (hsafe epsilon he)).2.le

/-- A concrete sequence implements the source's v_k -> infinity argument, with
actual boundary states, attained braking first-hit times, and exact x_k and
v_k/a arithmetic. It is independent of a reward convention. -/
theorem actual_unbounded_boundary_sequence_has_attained_first_hits_tending_to_infinity
    (a L : ℝ) (ha : 0 < a) :
    Tendsto (fun n : ℕ => a*((n+1:ℕ):ℝ)) atTop atTop ∧
      Tendsto (fun n : ℕ => ((n+1:ℕ):ℝ)) atTop atTop ∧
      ∀ n : ℕ,
        let T : ℝ := ((n+1:ℕ):ℝ)
        let x : ℝ := L-(a/2)*T^2
        let v : ℝ := a*T
        x = L-v^2/(2*a) ∧ v/a=T ∧ ¬ Viable a L x v ∧
          brakingPosition a x v T=L ∧
          (∀ t : ℝ, 0 ≤ t → t < T → brakingPosition a x v t < L) := by
  have hT : Tendsto (fun n : ℕ => ((n+1:ℕ):ℝ)) atTop atTop :=
    tendsto_natCast_atTop_atTop.comp (tendsto_add_atTop_nat 1)
  refine ⟨hT.const_mul_atTop ha,hT,?_⟩
  intro n
  have hp : 0 < ((n+1:ℕ):ℝ) := by positivity
  have hf := actual_boundary_braking_family_first_hit a L ((n+1:ℕ):ℝ) ha hp
  dsimp only
  refine ⟨?_,?_,hf.1,hf.2.1,hf.2.2⟩
  · field_simp <;> ring
  · field_simp
