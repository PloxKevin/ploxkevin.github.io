theorem actual_printed_loop_transform_signs_have_the_correct_nonsymmetric_qc
    {n : Type*} [Fintype n] [DecidableEq n]
    (M : Matrix n n ℝ) (hoff : ∀ i j, i≠j → M i j≤0)
    (hrow : ∀ i, 0≤∑ j, M i j) (hcol : ∀ j, 0≤∑ i, M i j)
    (phi : ℝ→ℝ) (mu nu : ℝ) (hs : CompleteModulesLipSDP.slopeRestricted phi mu nu)
    (hz : phi 0=0) (v : n→ℝ) :
    0 ≤ (fun i => mu*v i-phi (v i)) ⬝ᵥ
      (M *ᵥ (fun i => phi (v i)-nu*v i)) := by
  have ho : ∀ i j, i≠j → Mᵀ i j≤0 := by
    intro i j hij
    exact hoff j i hij.symm
  have hr : ∀ i, 0≤∑ j, Mᵀ i j := hcol
  have hc : ∀ j, 0≤∑ i, Mᵀ i j := hrow
  have h := actual_loop_transformed_slope_channel_has_the_full_monovary_qc
    Mᵀ ho hr hc phi mu nu hs hz v
  rw [Matrix.dotProduct_transpose_mulVec] at h
  have hv : (fun i => mu*v i-phi (v i)) = -(fun i => phi (v i)-mu*v i) := by
    funext i
    simp only [Pi.neg_apply]
    ring
  have hw : (fun i => phi (v i)-nu*v i) = -(fun i => nu*v i-phi (v i)) := by
    funext i
    simp only [Pi.neg_apply]
    ring
  rw [hv,hw,Matrix.mulVec_neg,neg_dotProduct_neg]
  exact h

theorem actual_printed_loop_transformed_delay_window_has_a_true_hard_fir_iqc
    (ell N : ℕ) (c : ℤ→ℝ) (hc : ∀ d, d≠0 → c d≤0)
    (hcs : 0≤∑ d∈Finset.Icc (-(ell:ℤ)) (ell:ℤ), c d)
    (phi : ℝ→ℝ) (mu nu : ℝ) (hs : CompleteModulesLipSDP.slopeRestricted phi mu nu)
    (hz : phi 0=0) (v : ℕ→ℝ) :
    0 ≤ ∑ t : Fin (N+1), actualFIRWindowSupply ell c
      (fun i => mu*v i-phi (v i)) (fun i => phi (v i)-nu*v i) t.val := by
  rw [actual_finite_causal_window_quadratic_sum_is_the_exact_banded_toeplitz_form]
  obtain ⟨ho,hr,hcol⟩ := CompleteModulesNNTemporalDiagonal25.
    actual_every_finite_fir_prefix_toeplitz_matrix_is_doubly_hyperdominant
      (Finset.Icc (-(ell:ℤ)) (ell:ℤ)) c hc hcs N
  have h := actual_printed_loop_transform_signs_have_the_correct_nonsymmetric_qc
    (CompleteModulesNNTemporalDiagonal25.prefixToeplitz
      (Finset.Icc (-(ell:ℤ)) (ell:ℤ)) c N) ho hr hcol phi mu nu hs hz
    (fun i => v i.val)
  have he : (fun i : Fin (N+1) => mu*v i.val-phi (v i.val)) ⬝ᵥ
      (CompleteModulesNNTemporalDiagonal25.prefixToeplitz
        (Finset.Icc (-(ell:ℤ)) (ell:ℤ)) c N *ᵥ
          (fun i => phi (v i.val)-nu*v i.val)) =
      ∑ i : Fin (N+1), ∑ j : Fin (N+1),
        (mu*v i.val-phi (v i.val))*CompleteModulesNNTemporalDiagonal25.prefixToeplitz
          (Finset.Icc (-(ell:ℤ)) (ell:ℤ)) c N i j*(phi (v j.val)-nu*v j.val) := by
    simp [dotProduct,Matrix.mulVec,Finset.mul_sum,mul_assoc]
  rw [he] at h
  exact mul_nonneg (by norm_num) h
