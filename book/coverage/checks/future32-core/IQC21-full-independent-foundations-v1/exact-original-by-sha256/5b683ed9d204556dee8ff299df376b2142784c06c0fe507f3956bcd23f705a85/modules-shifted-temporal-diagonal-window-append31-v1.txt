theorem actual_shifted_independent_channels_satisfy_the_printed_hard_delay_window_iqc
    {n : Type*} [Fintype n]
    (ell N : ℕ) (c : n→ℤ→ℝ) (hc : ∀ k d, d≠0 → c k d≤0)
    (hcs : ∀ k, 0≤∑ d∈Finset.Icc (-(ell:ℤ)) (ell:ℤ), c k d)
    (center : n→ℝ) (v : ℕ→n→ℝ) :
    0≤∑ k : n, ∑ t : Fin (N+1), actualFIRWindowSupply ell (c k)
      (fun t => -CompleteModulesNNTanhOffset24.shifted (center k) (v t k))
      (fun t => CompleteModulesNNTanhOffset24.shifted (center k) (v t k)-v t k) t.val := by
  apply Finset.sum_nonneg
  intro k _
  have h := CompleteModulesNNTanhOffset24.
    actual_every_shifted_channel_vanishes_and_has_own_valid_sector (center k)
  simpa only [zero_mul,one_mul,zero_sub] using
    actual_printed_loop_transformed_delay_window_has_a_true_hard_fir_iqc
      ell N (c k) (hc k) (hcs k) (CompleteModulesNNTanhOffset24.shifted (center k))
      0 1 h.2.1 h.1 (fun t => v t k)

theorem actual_independent_shifted_channel_filter_realization_has_the_full_middle_matrix_hard_iqc
    {n : Type*} [Fintype n]
    (ell N : ℕ) (c : n→ℤ→ℝ) (hc : ∀ k d, d≠0 → c k d≤0)
    (hcs : ∀ k, 0≤∑ d∈Finset.Icc (-(ell:ℤ)) (ell:ℤ), c k d)
    (center : n→ℝ) (v : ℕ→n→ℝ) :
    0≤∑ k : n, ∑ t : Fin (N+1),
      (Sum.elim
        (actualLagWindow ell (fun s => -CompleteModulesNNTanhOffset24.shifted (center k) (v s k)) t.val)
        (actualLagWindow ell (fun s => CompleteModulesNNTanhOffset24.shifted (center k) (v s k)-v s k) t.val)) ⬝ᵥ
      (actualFIRWindowMiddle ell (c k) *ᵥ
        (Sum.elim
          (actualLagWindow ell (fun s => -CompleteModulesNNTanhOffset24.shifted (center k) (v s k)) t.val)
          (actualLagWindow ell (fun s => CompleteModulesNNTanhOffset24.shifted (center k) (v s k)-v s k) t.val))) := by
  simpa only [actual_full_printed_FIR_window_quadratic_form_equals_its_exact_supply] using
    actual_shifted_independent_channels_satisfy_the_printed_hard_delay_window_iqc
      ell N c hc hcs center v

end SafeLearning.CompleteModulesShiftedTemporalIQC31
