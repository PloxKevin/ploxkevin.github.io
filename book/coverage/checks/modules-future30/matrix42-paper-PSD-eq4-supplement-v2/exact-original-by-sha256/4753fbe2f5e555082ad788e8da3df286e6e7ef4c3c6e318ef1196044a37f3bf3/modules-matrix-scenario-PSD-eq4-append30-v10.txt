def actualClosedUnitDisk (C : Matrix State State ℝ) : Prop :=
  ∀ z ∈ (C.map Complex.ofReal).charpoly.roots, ‖z‖ ≤ 1

def initialEqFourSemidefiniteCertificate (m : Plant) (Y Z : Matrix State State ℝ) (L : State → ℝ)
    (Qhalf noisehalf : Matrix State State ℝ) (r : ℝ) : Prop :=
  Y.PosDef ∧ 0 < r ∧ (initialEqFourMatrix m Y L Qhalf r).PosSemidef ∧
    (Matrix.fromBlocks Z noisehalf noisehalf.transpose Y).PosSemidef

section SemidefiniteEnergy
variable {n : Type*} [Fintype n] [DecidableEq n]

theorem actual_real_semidefinite_energy_is_nonnegative_on_every_complex_vector
    (P : Matrix n n ℝ) (hP : P.PosSemidef) (v : n → ℂ) :
    0 ≤ complexRealEnergy P v := by
  rw [actual_complex_energy_is_the_sum_of_real_and_imaginary_energies]
  have hr := hP.dotProduct_mulVec_nonneg (fun i => (v i).re)
  have hi := hP.dotProduct_mulVec_nonneg (fun i => (v i).im)
  simpa only [star_trivial] using add_nonneg hr hi

theorem actual_semidefinite_discrete_Lyapunov_drop_bounds_complex_eigenvalue_norm_by_one
    (P C : Matrix n n ℝ) (hP : P.PosDef) (hdrop : (P-C.transpose*P*C).PosSemidef)
    (v : n → ℂ) (hv : v ≠ 0) (z : ℂ)
    (heigen : C.map Complex.ofReal *ᵥ v=z • v) : ‖z‖ ≤ 1 := by
  have hp := actual_real_positive_definiteness_is_positive_on_every_nonzero_complex_vector P hP v hv
  have hd := actual_real_semidefinite_energy_is_nonnegative_on_every_complex_vector _ hdrop v
  rw [actual_discrete_Lyapunov_difference_is_the_complex_energy_drop,heigen,
    actual_complex_energy_scales_by_squared_eigenvalue_norm] at hd
  by_contra hn
  have hnorm : 1 < ‖z‖ := lt_of_not_ge hn
  have hsq : 1 < ‖z‖^2 := (one_lt_sq_iff₀ (norm_nonneg z)).mpr hnorm
  have hprod := mul_lt_mul_of_pos_right hsq hp
  nlinarith
end SemidefiniteEnergy

theorem actual_semidefinite_discrete_Lyapunov_certificate_has_closed_unit_disk_roots
    (P C : Matrix State State ℝ) (hP : P.PosDef) (hdrop : (P-C.transpose*P*C).PosSemidef) :
    actualClosedUnitDisk C := by
  intro z hz
  have hs : z ∈ spectrum ℂ (C.map Complex.ofReal) :=
    Matrix.mem_spectrum_iff_isRoot_charpoly.mpr (Polynomial.isRoot_of_mem_roots hz)
  have he : Module.End.HasEigenvalue (C.map Complex.ofReal).toLin' z :=
    Module.End.hasEigenvalue_iff_mem_spectrum.mpr (by simpa using hs)
  obtain ⟨v,hv⟩ := he.exists_hasEigenvector
  have hv' : C.map Complex.ofReal *ᵥ v=z • v := by simpa using hv.apply_eq_smul
  exact actual_semidefinite_discrete_Lyapunov_drop_bounds_complex_eigenvalue_norm_by_one
    P C hP hdrop v hv.2 z hv'

theorem actual_closed_unit_disk_four_state_matrix_has_trace_absolute_value_at_most_four
    (C : Matrix State State ℝ) (hC : actualClosedUnitDisk C) : |C.trace| ≤ 4 := by
  have hsum := actual_complex_root_sum_norm_is_bounded_by_multiplicity_card
    (C.map Complex.ofReal).charpoly.roots hC
  have hcard : ((C.map Complex.ofReal).charpoly.roots.card:ℝ) ≤ 4 := by
    have h := Polynomial.card_roots' (C.map Complex.ofReal).charpoly
    simp only [Matrix.charpoly_natDegree_eq_dim,Fintype.card_fin] at h
    exact_mod_cast h
  rw [←Matrix.trace_eq_sum_roots_charpoly] at hsum
  have he : (C.map Complex.ofReal).trace=(C.trace:ℂ) := by
    simp [Matrix.trace,Matrix.diag,Matrix.map_apply]
  rw [he,Complex.norm_real,Real.norm_eq_abs] at hsum
  exact hsum.trans hcard
theorem actual_small_trace_plant_unit_disk_roots_bound_the_corresponding_gain_coordinate
    (m : Plant) (i : State) (hm : m ∈ smallTracePlantNeighborhood i)
    (K : State → ℝ) (hs : actualClosedUnitDisk (closedLoop m K)) :
    |K i| ≤ 5+(1/64)*gainL1 K := by
  have hc := actual_closed_unit_disk_four_state_matrix_has_trace_absolute_value_at_most_four _ hs
  rw [actual_rank_one_state_feedback_trace_is_the_input_gain_pairing] at hc
  have he := actual_input_perturbation_pairing_error_is_uniform_in_the_gain m i hm.2 K
  have htri := abs_add_le (m.1.trace+∑ j,m.2 j*K j)
    (-m.1.trace-((∑ j,m.2 j*K j)-K i))
  have htri' := abs_add_le (-m.1.trace) (-((∑ j,m.2 j*K j)-K i))
  have hident : m.1.trace+(∑ j,m.2 j*K j)+
      (-m.1.trace-((∑ j,m.2 j*K j)-K i))=K i := by ring
  rw [hident] at htri
  simp only [abs_neg] at htri'
  change |m.1.trace| < 1 ∧ inputClose m i at hm
  simp only [sub_eq_add_neg] at htri htri' he
  linarith [hm.1]

theorem actual_four_small_trace_unit_disk_models_bound_the_entire_common_gain
    (s : State → Plant) (hs : ∀ i,s i ∈ smallTracePlantNeighborhood i)
    (K : State → ℝ) (hstable : ∀ i,actualClosedUnitDisk (closedLoop (s i) K)) :
    gainL1 K ≤ 64/3 := by
  have hb := Finset.sum_le_sum (fun (i : State) (_ : i ∈ Finset.univ) =>
    actual_small_trace_plant_unit_disk_roots_bound_the_corresponding_gain_coordinate
      (s i) i (hs i) K (hstable i))
  simp only [Finset.sum_const,Finset.card_univ,Fintype.card_fin,nsmul_eq_mul] at hb
  change gainL1 K ≤ 4*(5+(1/64)*gainL1 K) at hb
  linarith

theorem actual_open_five_model_neighborhood_has_no_common_closed_unit_disk_gain
    (s : Fin 5 → Plant) (hs : s ∈ incompatibleFivePlantNeighborhood) :
    ¬ ∃ K : State → ℝ, ∀ i,actualClosedUnitDisk (closedLoop (s i) K) := by
  rintro ⟨K,hK⟩
  have hL := actual_four_small_trace_unit_disk_models_bound_the_entire_common_gain
    (fun i => s i.castSucc) hs.1 K (fun i => hK i.castSucc)
  have hcoordinate := actual_small_trace_plant_unit_disk_roots_bound_the_corresponding_gain_coordinate
    (s (3:State).castSucc) 3 (hs.1 3) K (hK (3:State).castSucc)
  have he := actual_input_perturbation_pairing_error_is_uniform_in_the_gain
    (s (Fin.last 4)) 3 hs.2.2 K
  have hc := actual_closed_unit_disk_four_state_matrix_has_trace_absolute_value_at_most_four _
    (hK (Fin.last 4))
  rw [actual_rank_one_state_feedback_trace_is_the_input_gain_pairing] at hc
  have hlow : -(5+(2/64)*gainL1 K) ≤ ∑ j,(s (Fin.last 4)).2 j*K j := by
    have hcoord := (abs_le.mp hcoordinate).1
    have herr := (abs_le.mp he).1
    linarith
  have hupper := (abs_le.mp hc).2
  have hlast : 11 < (s (Fin.last 4)).1.trace ∧ inputClose (s (Fin.last 4)) 3 := hs.2
  linarith [hlast.1]

theorem actual_full_semidefinite_eq_four_LMI_has_the_true_semidefinite_principal_block
    (m : Plant) (Y : Matrix State State ℝ) (L : State → ℝ)
    (Qhalf : Matrix State State ℝ) (r : ℝ)
    (hM : (initialEqFourMatrix m Y L Qhalf r).PosSemidef) :
    (initialEqFourPrincipal m Y L).PosSemidef := by
  have h := hM.submatrix (fun x : State ⊕ State =>
    (Sum.inl x : (State ⊕ State) ⊕ (State ⊕ Fin 1)))
  convert h using 1
  ext i j
  rfl

theorem actual_semidefinite_eq_four_initial_LMI_recovers_closed_unit_disk_gain
    (m : Plant) (Y : Matrix State State ℝ) (hY : Y.PosDef) (L : State → ℝ)
    (Qhalf : Matrix State State ℝ) (r : ℝ)
    (hM : (initialEqFourMatrix m Y L Qhalf r).PosSemidef) :
    actualClosedUnitDisk (closedLoop m (L ᵥ* Y⁻¹)) := by
  letI := hY.isUnit.invertible
  let C := closedLoop m (L ᵥ* Y⁻¹)
  let W := m.1*Y+Matrix.vecMulVec m.2 L
  have hw : W=C*Y := (actual_L_equals_KY_recovers_the_true_rank_one_closed_loop m Y hY L).symm
  have hy : Y.transpose=Y := by
    simpa only [Matrix.conjTranspose_eq_transpose_of_trivial] using hY.isHermitian.eq
  have hh := actual_full_semidefinite_eq_four_LMI_has_the_true_semidefinite_principal_block m Y L Qhalf r hM
  change (Matrix.fromBlocks Y W.transpose W Y).PosSemidef at hh
  have hd : (Y-W*Y⁻¹*W.transpose).PosSemidef := by
    have h := (Matrix.PosDef.fromBlocks₁₁ W.transpose Y hY).mp
      (by simpa only [Matrix.conjTranspose_eq_transpose_of_trivial,Matrix.transpose_transpose] using hh)
    simpa only [Matrix.conjTranspose_eq_transpose_of_trivial,Matrix.transpose_transpose] using h
  have he : W*Y⁻¹*W.transpose=C*Y*C.transpose := by
    rw [hw,Matrix.transpose_mul,hy,Matrix.mul_assoc C Y Y⁻¹,
      Matrix.mul_inv_of_invertible,Matrix.mul_one,←Matrix.mul_assoc]
  rw [he] at hd
  have hs : actualClosedUnitDisk C.transpose :=
    actual_semidefinite_discrete_Lyapunov_certificate_has_closed_unit_disk_roots
      Y C.transpose hY (by simpa only [Matrix.transpose_transpose] using hd)
  intro z hz
  exact hs z (by simpa only [Matrix.transpose_map,Matrix.charpoly_transpose] using hz)

theorem actual_full_semidefinite_eq_four_certificate_gives_closed_unit_disk_gain
    (m : Plant) (Y Z : Matrix State State ℝ) (L : State → ℝ)

    (h : initialEqFourSemidefiniteCertificate m Y Z L Qhalf noisehalf r) :
    actualClosedUnitDisk (closedLoop m (L ᵥ* Y⁻¹)) :=
  actual_semidefinite_eq_four_initial_LMI_recovers_closed_unit_disk_gain m Y h.1 L Qhalf r h.2.2.1

theorem actual_open_five_model_event_makes_the_paper_semidefinite_initial_LMI_infeasible
    (s : Fin 5 → Plant) (hs : s ∈ incompatibleFivePlantNeighborhood)
    (Qhalf noisehalf : Matrix State State ℝ) (r : ℝ) :
    ¬ ∃ (Y Z : Matrix State State ℝ) (L : State → ℝ),
      ∀ i,initialEqFourSemidefiniteCertificate (s i) Y Z L Qhalf noisehalf r := by
  rintro ⟨Y,Z,L,h⟩
  apply actual_open_five_model_neighborhood_has_no_common_closed_unit_disk_gain s hs
  exact ⟨L ᵥ* Y⁻¹,fun i => actual_full_semidefinite_eq_four_certificate_gives_closed_unit_disk_gain
    (s i) Y Z L Qhalf noisehalf r (h i)⟩
