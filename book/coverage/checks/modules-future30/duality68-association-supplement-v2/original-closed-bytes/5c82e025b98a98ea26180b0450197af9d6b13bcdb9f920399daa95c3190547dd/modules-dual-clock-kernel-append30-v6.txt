def actualClockViable (s : Option ℕ) : Prop :=
  ∃v : ℕ→Bool,∀t,feedbackTrajectory s v t≠none

theorem actual_clock_viability_kernel_and_uniform_immediate_post_exit_failure :
    {s : Option ℕ | actualClockViable s}={s | s≠none} ∧
      ∀s,¬actualClockViable s→∀v,∃t≤0,feedbackTrajectory s v t=none := by
  have hk : ∀s : Option ℕ,actualClockViable s↔s≠none := by
    intro s
    cases s with
    | none =>
      constructor
      · rintro ⟨v,hv⟩
        exact False.elim (hv 0 rfl)
      · intro h;exact False.elim (h rfl)
    | some n =>
      constructor
      · intro _;simp
      · intro _
        refine ⟨fun _=>false,?_⟩
        intro t
        rw [actual_sink_is_immediate_failure_and_all_surviving_clocks_have_a_safe_feedback.2 n t]
        simp
  constructor
  · ext s;exact hk s
  · intro s hs v
    have he : s=none := by
      by_contra hn
      exact hs ((hk s).mpr hn)
    subst s
    exact ⟨0,le_rfl,rfl⟩

end SafeLearning.CompleteModulesViabilityDualClock30
