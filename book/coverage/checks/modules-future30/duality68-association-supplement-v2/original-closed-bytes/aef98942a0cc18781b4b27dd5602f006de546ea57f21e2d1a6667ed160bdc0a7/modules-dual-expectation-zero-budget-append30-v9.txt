def actualZeroBudgetPrimal {U : Type*} (reward risk : U→ℝ) : ℝ :=
  ⨆u : {u : U // risk u≤0},reward u.val

theorem actual_nonnegative_risk_zero_budget_CMDP_is_the_actual_zero_risk_primal
    {U : Type*} (reward risk : U→ℝ) (hn : ∀u,0≤risk u) :
    actualZeroBudgetPrimal reward risk=actualPrimal reward risk := by
  have he : (fun u : U=>risk u≤0)=(fun u : U=>risk u=0) := by
    funext u
    apply propext
    exact ⟨fun h=>le_antisymm h (hn u),fun h=>h.le⟩
  simp only [actualZeroBudgetPrimal,actualPrimal,he]

theorem actual_zero_budget_CMDP_feasibility_is_exactly_almost_sure_safety
    {X U : Type*} [MeasurableSpace X] (mu : Measure X) (rho : X→U→ℝ) (u : U)
    (hi : Integrable (fun x=>rho x u) mu) (hn : ∀x,0≤rho x u) :
    expectedRisk mu rho u≤0 ↔ ∀ᵐx ∂mu,rho x u=0 := by
  have he : expectedRisk mu rho u≤0 ↔ expectedRisk mu rho u=0 :=
    ⟨fun h=>le_antisymm h (integral_nonneg hn),fun h=>h.le⟩
  exact he.trans (actual_expected_zero_risk_iff_almost_sure_zero_pointwise_risk mu rho u hi hn)

def projectedBudgetStep (lambda eta cost budget : ℝ) : ℝ :=
  max 0 (lambda+eta*(cost-budget))

theorem actual_positive_budget_zero_cost_exact_maximizer_process_does_not_oscillate
    (eta budget : ℝ) (heta : 0<eta) (hbudget : 0<budget) :
    (∀p : ℝ,∀u v : Fin 1,(0:ℝ)-p*0≤0-p*0) ∧
      (∀n : ℕ,projectedBudgetStep 0 eta 0 budget=0) ∧
      (∀n : ℕ,|((fun _ : ℕ=>(0:ℝ)) (n+1))-(fun _ : ℕ=>(0:ℝ)) n|=0) := by
  refine ⟨by intros;norm_num,?_,by intros;norm_num⟩
  intro n
  apply max_eq_left
  dsimp
  nlinarith
